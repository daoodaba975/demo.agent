# Le code de <span class="accent">notre agent</span>

<div class="grid grid-cols-5 gap-4 mt-2">
<div class="col-span-3">

```python {1-5|7-11|13-15|16|all}
@tool  # 1. L'outil (la main)
def lire_notes() -> str:
    """Lit le fichier notes-cours.txt et renvoie son contenu."""
    with open("notes-cours.txt", encoding="utf-8") as f:
        return f.read()

model = OpenAIModel(  # 2. Le modèle (le cerveau)
    model_id=os.getenv("MODEL_ID"),
    api_base=os.getenv("API_BASE"),
    api_key=os.getenv("API_KEY"),  # la clé vient du .env
)

consignes = "Tu es Jang, un assistant qui aide des étudiants..."
agent = ToolCallingAgent(tools=[lire_notes], model=model,
                         instructions=consignes)  # 3. Consignes + agent
print(agent.run(sys.argv[1]))  # 4. La boucle
```

</div>
<div class="col-span-2 flex flex-col gap-3">
  <div class="card">
    <div class="font-bold accent text-sm mb-1">Outil</div>
    <div class="text-xs muted"><code>lire_notes</code> : une simple fonction Python. La docstring explique au modèle à quoi elle sert.</div>
  </div>
  <div class="card">
    <div class="font-bold accent text-sm mb-1">Modèle + consignes</div>
    <div class="text-xs muted">Le modèle est choisi dans le fichier <code>.env</code>. Les consignes lui donnent son rôle.</div>
  </div>
  <div class="card">
    <div class="font-bold accent text-sm mb-1">Boucle</div>
    <div class="text-xs muted"><code>agent.run()</code> : réfléchir, appeler l'outil, observer, répondre. 29 lignes en tout.</div>
  </div>
</div>
</div>
